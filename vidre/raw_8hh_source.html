543 inline const std::optional<std::endian> check_magic(uint64_t m) {
544 if (m == magic) {
545 return std::endian::native;
546 }
547 else if (m == std::byteswap(magic)) {
548 return std::endian::native == std::endian::little ? std::endian::big : std::endian::little;
549 }
550 else
551 return {};
552 }
