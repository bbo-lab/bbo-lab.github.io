Coding style

C++

For C++, the coding style is provided in form of a .clang-format file. The major rules are:

  • indentation of 4 space, no tabs,
  • maximum width is 132
  • preprocessor statements are not indented
  • namespaces indent
  • braces are in line
  • etc.

void style::ExampleClass::processData(const std::vector<int> &values) {
for (const auto &value : values) {
if (checkCondition(value)) {
cached_data_.push_back(value);
}
else {
std::cout << "Skipping invalid value: " << value << std::endl;
}
}
validateInternal();
}
bool style::ExampleClass::isValid() const noexcept {
return !name_.empty() && !cached_data_.empty();
}
void style::ExampleClass::validateInternal() const {
if (cached_data_.empty()) {
throw std::runtime_error("No valid data to process");
}
}
bool style::ExampleClass::checkCondition(int value) const {
return value > 0 && value < 100;
