239 class VIDRE_CORE_VISIBILITY store {
240 public:
241 using size_type = io::size_type;
242 using offset_type = io::offset_type;
243
244 static constexpr size_type alignment = io::packet::alignment;
245 static constexpr size_type size = VIDRE_CORE_LOG_BUFFER_SIZE;
246
247 public:
248 store() = default;
249 store(const store &) = delete;
250 store &operator=(const store &) = delete;
251 store(store &&) = delete;
252 store &operator=(store &&) = delete;
253
254 protected:
255 static thread_local char buffer_[size];
256
257 static thread_local std::mutex mutex_;
258 static thread_local io::size_type begin_;
259 static thread_local io::size_type end_;
260 static thread_local io::size_type used_;
261
262 private:
265 static void flush() noexcept {
266 std::lock_guard<std::mutex> lock(mutex_);
267
268 if (used_ > end_)
269 return;
270
271 if ((begin_ == end_) and (begin_ > 0)) {
272 used_ = 0;
273 end_ = begin_ = 0;
274 }
275 }
276
277 public:
281 static size_type free() noexcept {
282 return size - used_;
283 }
284
290 static lognew::packet *get() noexcept {
291 std::lock_guard<std::mutex> lock(mutex_);
292
293 if (begin_ >= end_)
294 return nullptr;
295
296 lognew::packet *pkt = reinterpret_cast<lognew::packet *>(buffer_ + begin_);
297 return pkt;
298 }
299
304 static void pop() noexcept {
305 std::lock_guard<std::mutex> lock(mutex_);
306
307 if (begin_ >= end_)
308 return;
309
310 lognew::packet *pkt = reinterpret_cast<lognew::packet *>(buffer_ + begin_);
311 begin_ += sizeof(io::packet::basic) + pkt->size();
312 }
313
319 static lognew::packet *generate(level lvl, uint16_t stream = 0) noexcept {
320 flush();
321
322 if (free() < sizeof(lognew::packet))
323 // should not happen...
324 return nullptr;
325
326 lognew::packet *ptr = reinterpret_cast<lognew::packet *>(buffer_ + used_);
327 new (ptr) lognew::packet(lvl, stream);
328
329 used_ += sizeof(lognew::packet);
330 return ptr;
331 }
332
335 static void append(const std::string_view &sv) noexcept {
336 if (used_ == end_)
337 // should not happen...
338 return;
339
340 auto n = std::min(free(), sv.size()); // ignore, if there's not enough space
341
342 std::memcpy(buffer_ + used_, sv.data(), n);
343 used_ += n;
344
345 lognew::packet *pkt = reinterpret_cast<lognew::packet *>(buffer_ + begin_);
346 pkt->length += n;
347 }
348
351 static void finalize() {
352 lognew::packet *pkt = reinterpret_cast<lognew::packet *>(buffer_ + begin_);
353
354 size_t padding = pkt->size() - pkt->length;
355 if (padding > 0) {
356 static const uint8_t padbuf[io::packet::alignment] = {};
357 std::memcpy(buffer_ + used_, padbuf, padding);
358 used_ += padding;
359 }
360
361 {
362 std::lock_guard<std::mutex> lock(mutex_);
363 end_ = used_;
364 }
365
366 // FIXME could notify log processing thread here
367 // (current implementation just cycles through buffers)
368 }
369 };
