54 class VIDRE_DEVICE_VISIBILITY state {
55 public:
56 enum flag : uint32_t {
57 undefined = 0,
58 available = 1 << 0,
59 disabled = 1 << 1,
60 error = 1 << 2,
61 connected = 1 << 3,
62 configured = 1 << 4,
63 running = 1 << 5,
64 };

66 template<typename T>
67 static constexpr bool compatible_v = std::derived_from<T, state> or std::derived_from<T, state::flag>;
68
69 state() noexcept : flags_(flag::undefined) {}
70 state(flag f) noexcept : flags_(f) {}
71
72 protected:
73 uint32_t flags_;
74
75 public:
76 operator bool() const noexcept {
77 return flags_ != static_cast<uint32_t>(flag::undefined);
78 }
79
80 bool operator!() const noexcept {
81 return flags_ == static_cast<uint32_t>(flag::undefined);
82 }
83
84 bool operator==(flag f) const noexcept {
85 return flags_ == static_cast<uint32_t>(f);
86 }
87
88 bool operator==(state s) const noexcept {
89 return flags_ == s.flags_;
90 }
91
92 bool has(flag f) const noexcept {
93 return (flags_ & static_cast<uint32_t>(f)) != 0;
94 }
95
96 state &set(flag f) noexcept {
97 flags_ |= static_cast<uint32_t>(f);
98 return *this;
99 }
100
101 state &clear(flag f) noexcept {
102 flags_ &= ~static_cast<uint32_t>(f);
103 return *this;
104 }
105
106 state &toggle(flag f) noexcept {
107 flags_ ^= static_cast<uint32_t>(f);
108 return *this;
109 }
110
111 state &reset() noexcept {
112 flags_ = static_cast<uint32_t>(flag::undefined);
113 return *this;
114 }
115
116 state &operator&=(flag f) noexcept {
117 flags_ &= static_cast<uint32_t>(f);
118 return *this;
119 }
120
121 state &operator&=(const state &f) noexcept {
122 flags_ &= f.flags_;
123 return *this;
124 }
125
126 state &operator|=(flag f) noexcept {
127 flags_ |= static_cast<uint32_t>(f);
128 return *this;
129 }
130
131 state &operator|=(const state &f) noexcept {
132 flags_ |= f.flags_;
133 return *this;
134 }
135 };
