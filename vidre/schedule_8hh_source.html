80 static int min_priority(policy_type policy_) noexcept {
81 switch (policy_) {
82 case policy_type::system:
83 case policy_type::batch:
84 // For SCHED_OTHER/SCHED_BATCH, "priority" is really nice value
85 // Min priority = 19 (lowest priority, highest nice)
86 return 19;
87
88 case policy_type::fifo:
89 case policy_type::round_robin:
90 return sched_get_priority_min(static_cast<int>(policy_));
91
92 case policy_type::deadline:
93 // SCHED_DEADLINE uses runtime/deadline/period, not traditional priority
94 // Return 0 as placeholder - actual priority is determined by parameters
95 return 0;
96
97 case policy_type::idle:
98 // SCHED_IDLE has no priority values - always runs at lowest priority
99 return 0;
100
101 default:
102 return -1;
103 }
104 }
105
110 static int max_priority(policy_type policy_) noexcept {
111 switch (policy_) {
112 case policy_type::system:
113 case policy_type::batch:
114 // For SCHED_OTHER/SCHED_BATCH, max priority = -20 (highest priority, lowest nice)
115 return -20;
116
117 case policy_type::fifo:
118 case policy_type::round_robin:
119 return sched_get_priority_max(static_cast<int>(policy_));
120
121 case policy_type::deadline:
122 // SCHED_DEADLINE returns 0 as placeholder - use sched_setattr for real config
123 return 0;
124
125 case policy_type::idle:
126 return 0;
127
128 default:
129 return -1;
130 }
131 }

138 static int normalized_priority(float normalized, policy_type policy_) noexcept {
139 if (normalized < 0.0f)
140 normalized = 0.0f;
141 if (normalized > 1.0f)
142 normalized = 1.0f;
143
144 int min_prio = min_priority(policy_);
145 int max_prio = max_priority(policy_);
146
147 if (min_prio == max_prio)
148 return min_prio;
149
150 switch (policy_) {
151 case policy_type::system:
152 case policy_type::batch:
153 // Nice values: lower = higher priority
154 // Normalized 1.0 (highest) -> max_prio (-20)
155 // Normalized 0.0 (lowest) -> min_prio (19)
156 return static_cast<int>(max_prio - (normalized * (max_prio - min_prio)));
157
158 case policy_type::fifo:
159 case policy_type::round_robin:
160 // SCHED_FIFO/RR: higher = higher priority
161 // Normalized 1.0 -> max_prio
162 // Normalized 0.0 -> min_prio
163 return static_cast<int>(min_prio + (normalized * (max_prio - min_prio)));
164
165 default:
166 return min_prio;
167 }
168 }
