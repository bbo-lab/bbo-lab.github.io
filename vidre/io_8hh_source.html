552 struct VIDRE_CORE_VISIBILITY buffer {
553 using size_type = std::size_t;
554 using value_type = uint8_t;
555 using pointer = value_type *;
556 using const_pointer = const value_type *;
557
558 buffer() = default;
559 ~buffer() {
560 if (data)
561 std::free(data);
562 }
563
564 size_type size{};
565 size_type used{};
566 pointer data;
567
578 size_type setup(size_type len) noexcept {
579 if (data)
580 return 0;
581
582 len = std::bit_ceil(len);
583 data = reinterpret_cast<pointer>(std::aligned_alloc(8, len));
584 if (data)
585 size = len;
586
587 return size;
588 }
589
592 operator bool() const noexcept {
593 return nullptr != data;
594 }
595
598 bool full() const noexcept {
599 return used == size;
600 }
601
606 size_type flush() noexcept {
607 size_type retval = used;
608 used = 0;
609 return retval;
610 }
611
617 size_type append(const_pointer src, size_type n) noexcept {
618 assert(data != nullptr);
619
620 size_type space = size - used;
621 size_type done = std::min(space, n);
622 std::memcpy(data + used, src, done);
623 used += done;
624 return done;
625 }
626 };

640 class VIDRE_CORE_VISIBILITY sink {
641 public:
642 using size_type = std::size_t;
643 using offset_type = std::make_signed_t<size_type>;
644
645 public:
646 sink() = default;
647 virtual ~sink() {}
648
649 protected:
650 buffer buf_{};
651
652 protected:
657 offset_type flush() {
658 if (buf_.used) {
659 auto retval = write(buf_.data, buf_.used);
660
661 if (retval != buf_.used)
662 // if the underlying hardware can't take it, we assume,
663 // it's out of memory...
664 throw exception::memory();
665 }
666 return buf_.flush();
667 }
668
672 virtual offset_type write(const uint8_t *buf, size_type count) = 0;
673
674 public:
677 virtual bool valid() const noexcept {
678 return false;
679 }
680
684 offset_type put(const uint8_t *buf, size_type count) {
685 if (buf_) {
686 // using buffer
687 offset_type retval = count;
688 offset_type pos = 0;
689 while (count > 0) {
690 size_type buffered = buf_.append(buf + pos, count);
691 if (buf_.full())
692 flush();
693 count -= buffered;
694 pos += buffered;
695 }
696
697 return retval;
698 }
699 else
700 // direct write
701 return write(buf, count);
702 }
703
704 offset_type put(const view &v) {
705 return put(v.data(), v.size());
706 }
707
708 template <typename T>
709 offset_type put(const T &data) {
710 return put(view{sizeof(T), &data});
711 }
712
713 template <typename T>
714 offset_type put(const T *data, size_type nmemb) {
715 return put(view{nmemb * sizeof(T), reinterpret_cast<const uint8_t *>(data)});
716 }

773 bool open(const std::filesystem::path &filename, mode_type mode = synchronous,
774 fileio::perms permissions = fileio::default_perms) noexcept {
775 fileio::flag_type f = fileio::writeonly | fileio::create | fileio::close_on_exec;
776
777 if (mode & mode_type::truncate)
778 f |= fileio::truncate;
779 else
780 f |= fileio::append;
781
782 if (mode & mode_type::synchronous)
783 f |= fileio::synchronous;
784
785 bool retval = fd_.open(filename, f, permissions);
786 if (not retval)
787 return false;
788
789 if ((mode & mode_type::synchronous) and not(mode & mode_type::unbuffered))
790 buf_.setup(detail::file::blocksize(fd_));
791
792 return true;
793 }
794
795 // FIXME setbuf(size_type n)
796
799 void close() noexcept {
800 if (buf_)
801 flush();
802 fd_.close();
803 return;
804 }
