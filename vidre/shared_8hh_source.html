144 template <typename T, typename Allocator = std::allocator<T>>
145 class shared {
146 public:
147 using allocator_type = Allocator;
148
149 private:
150 struct counted_object {
151 std::atomic<size_t> strong_count;
152 std::atomic<size_t> weak_count; // currently unused, would allow a weak implementation
153 T *object;
154
155 counted_object(T *ptr) : strong_count(1), weak_count(0), object(ptr) {}
156 };
157
158 using CountedObjectAllocator = typename std::allocator_traits<Allocator>::template rebind_alloc<counted_object>;
159
160 counted_object *object_{};
161
162 Allocator allocator{};
163 std::function<void(T *)> deleter;
164
165 private:
166 // Helper to destroy the counted object
167 void destroy_counted_object() noexcept {
168 if (object_) {
169 // Manually destroy the object and counted object
170 deleter(object_->object);
171 object_->~counted_object();
172
173 // Deallocate memory using the allocator
174 CountedObjectAllocator counted_object_allocator(allocator);
175 counted_object_allocator.deallocate(object_, 1);
176 }
177 }
178
179 void release() noexcept {
180 if (!object_)
181 return;
182
183 // Decrement strong count
184 size_t old_count = object_->strong_count.fetch_sub(1, std::memory_order_acq_rel);
185
186 if (old_count == 1) {
187 // No more strong references, destroy the object
188 destroy_counted_object();
189 }
190 }

192 public:
193 shared() noexcept {}
194
195 shared(T *ptr) noexcept {
196 deleter = [allocator = allocator](T *ptr) mutable {
197 ptr->~T();
198 allocator.deallocate(ptr, 1);
199 };
200
201 CountedObjectAllocator counted_object_allocator(allocator);
202 object_ = counted_object_allocator.allocate(1);
203
204 new (object_) counted_object(ptr);
205 }
