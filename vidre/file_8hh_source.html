169 inline std::string filename(descriptor fd) {
170#if defined(F_GETPATH)
171 char path[PATH_MAX];
172 if (fcntl(fd, F_GETPATH, path) != -1)
173 return std::string(path);
174 return {};
175#elif defined(__linux__)
176 char path[PATH_MAX];
177 ssize_t len = readlink((std::string("/proc/self/fd/") + std::to_string(fd)).c_str(), path, sizeof(path) - 1);
178 if (len != -1) {
179 path[len] = 0;
180 return std::string(path);
181 }
182 return {};
183#else
184 static_assert(false, "filename retrieval not implemented for this platform");
185 return {};
186#endif
187 }
