173 inline std::string filename(descriptor fd) {
174#if defined(F_GETPATH)
175 char path[PATH_MAX];
176 if (fcntl(fd, F_GETPATH, path) != -1)
177 return std::string(path);
178 return {};
179#elif defined(__linux__)
180 char path[PATH_MAX];
181 ssize_t len = readlink((std::string("/proc/self/fd/") + std::to_string(fd)).c_str(), path, sizeof(path) - 1);
182 if (len != -1) {
183 path[len] = 0;
184 return std::string(path);
185 }
186 return {};
187#else
188 static_assert(false, "filename retrieval not implemented for this platform");
189 return {};
190#endif
191 }
