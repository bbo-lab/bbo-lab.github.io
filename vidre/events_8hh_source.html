145 void terminate(std::chrono::milliseconds timeout) {
146 quit = true;
147
148 // clear the eventfd counter to block any waiter
149 value_type count;
150 read(handle, &count, sizeof(count));
151
152 // wake up any waiter to let them check the termination flag
153 if (notify(max_count) != success)
154 goto termination_signal_not_acknowledged;
155
156 // wait until we can write again (i.e. the waiter has acknowledged the termination signal)
157 {
158 struct pollfd pfd = {handle, POLLOUT, 0};
159 int ret = poll(&pfd, 1, static_cast<int>(timeout.count()));
160 if (ret > 0)
161 return; // acknowledged
162 }
163
164 termination_signal_not_acknowledged:
165 throw vidre::exception::runtime() << error::system() << error::message("failed to notify termination");
166 }

176 int notify(value_type count = 1) {
177 ssize_t ret = write(handle, &count, sizeof(count));
178 if (ret != sizeof(count)) {
179 switch (errno) {
180 case EAGAIN:
181 return try_again;
182 default:
183 throw vidre::exception::runtime() << error::system();
184 }
185 }
186 return success;
187 }

193 int avail(std::chrono::milliseconds timeout = {}) const noexcept {
194 struct pollfd pfd = {handle, POLLIN, 0};
195 return poll(&pfd, 1, static_cast<int>(timeout.count()));
196 }

201 result_type wait() {
202 value_type count;
203 ssize_t ret = read(handle, &count, sizeof(count));
204 while ((ret != sizeof(count)) and (errno == EAGAIN)) {
205 if (quit)
206 return {terminated, 0};
207 std::this_thread::yield();
208 ret = read(handle, &count, sizeof(count));
209 }
210 if (ret != sizeof(count)) {
211 switch (errno) {
212 case EAGAIN:
213 return {try_again, 0};
214 default:
215 throw vidre::exception::runtime() << error::system();
216 }
217 }
218 return {quit ? terminated : success, count};
219 }
220
225 result_type wait(std::chrono::milliseconds timeout) {
226 int ret = avail(timeout);
227 if (ret == 0)
228 return {timed_out, 0}; // timed out
229 else if (ret < 0) [[unlikely]] {
230 throw vidre::exception::runtime() << error::system();
231 }
