43#define VIDRE_PIXEL_FORMAT(NUM, COLOR, PACKED, BITS) ((NUM << 8) | ((!!COLOR) << 7) | ((!!PACKED) << 6) | (BITS & 0x7F))
44
45#define VIDRE_PIXEL_FORMAT_NUMBER(fmt) ((fmt >> 8) & 0xFFFFFF)
46#define VIDRE_PIXEL_FORMAT_COLOR(fmt) ((fmt >> 7) & 0x1)
47#define VIDRE_PIXEL_FORMAT_PACKED(fmt) ((fmt >> 6) & 0x1)
48#define VIDRE_PIXEL_FORMAT_BITS(fmt) (fmt & 0x3F)
