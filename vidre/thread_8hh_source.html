124 status priority(int priority) {
125 int policy;
126 sched_param sch_params;
127
128 // get current scheduling policy and parameters
129 if (pthread_getschedparam(handle, &policy, &sch_params) != 0) {
130 log::debug() << "Failed to get thread priority: " << std::strerror(errno) << log::endl;
131 return status::bad;
132 }
133
134 auto min_prio = min_priority(static_cast<policy_type>(policy));
135 auto max_prio = max_priority(static_cast<policy_type>(policy));
136
137 if ((priority < min_prio) || (priority > max_prio)) {
138 log::debug() << "Invalid priority value: " << priority << ". Valid range for policy "
139 << static_cast<int>(policy) << " is [" << min_prio << ", " << max_prio << "]" << log::endl;
140 return status::bad;
141 }
142
143 // set new priority
144 sch_params.sched_priority = priority;
145 if (pthread_setschedparam(handle, static_cast<int>(policy), &sch_params) != 0) {
146 log::debug() << "Failed to set thread priority: " << std::strerror(errno) << log::endl;
147 return status::bad;
148 }
149
150 return status::good;
151 }
