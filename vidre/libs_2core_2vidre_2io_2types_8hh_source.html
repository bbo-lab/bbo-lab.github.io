102 size_type setup(size_type len) noexcept {
103 if (data)
104 return 0;
105
106 len = std::bit_ceil(len);
107 data = reinterpret_cast<pointer>(std::aligned_alloc(8, len));
108 if (data)
109 size = len;
110
111 return size;
112 }
113
116 operator bool() const noexcept {
117 return nullptr != data;
118 }
119
122 bool full() const noexcept {
123 return used == size;
124 }
125
130 size_type flush() noexcept {
131 size_type retval = used;
132 used = 0;
133 return retval;
134 }
135
141 size_type append(const_pointer src, size_type n) noexcept {
142 assert(data != nullptr);
143
144 size_type space = size - used;
145 size_type done = std::min(space, n);
146 std::memcpy(data + used, src, done);
147 used += done;
148 return done;
149 }
