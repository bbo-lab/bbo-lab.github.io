100 buffer(size_type n) noexcept {

106 std::free(data);

109 size_type size{};
110 size_type used{};
111 pointer data{};
112
120 size_type setup(size_type n) noexcept {
121 if (data)
122 return 0;
123
124 n = std::bit_ceil(n);
125 data = reinterpret_cast<pointer>(std::aligned_alloc(8, n));
126 if (data)
127 size = n;
128
129 return size;
130 }
131
134 operator bool() const noexcept {
135 return nullptr != data;
136 }
137
140 bool full() const noexcept {
141 return used == size;
142 }
143
148 size_type flush() noexcept {
149 size_type retval = used;
150 used = 0;
151 return retval;
152 }
153
159 size_type put(const_pointer src, size_type n) noexcept {
160 assert(data != nullptr);
161
162 size_type space = size - used;
163 size_type done = std::min(space, n);
164 std::memcpy(data + used, src, done);
165 used += done;
166 return done;
167 }
168
171 operator memory::area() const noexcept {
172 return {size, data};
173 }
174 };
