144 // Allocate memory
145 T *allocate(std::size_t n) {
146 if (n > max_size()) {
147 throw std::bad_alloc();
148 }
149
150 // Call your static memory allocation function
151 void *ptr = store::allocate(n * sizeof(T));
152 if (!ptr) {
153 throw std::bad_alloc();

166 return static_cast<std::size_t>(-1) / sizeof(T);

305 template <class Info>
306 requires std::derived_from<Info, detail::exception::supplement>
307 void attach(Info &&inf) {
308 if (next_.get()) {
309 auto nextptr = dynamic_cast<info *>(next_.get());
310 nextptr->attach(std::move(inf));
311 }
312 else {
313 using Allocator = typename std::allocator_traits<shared_type::allocator_type>::template rebind_alloc<Info>;

316 Info *ptr = allocator.allocate(1);

318
319 next_ = shared_type(ptr, detail::exception::deleter<Info>());
320 }
